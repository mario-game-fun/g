fetch('https://data.classicgames.me/mario/media/mario-kart-circuit.swf')
  .then(response => response.blob())
  .then(blob => {
      const url = window.URL.createObjectURL(blob);
      const a = document.createElement('a');
      a.href = url;
      a.download = 'mario-kart-racing.swf';
      document.body.appendChild(a);
      a.click();
      a.remove();
  });
